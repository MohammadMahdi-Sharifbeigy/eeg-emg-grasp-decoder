\lr{v2} از سه نظر بهتر از مدل پایه است. اول، دقت: \lr{K} در هر ۱۲ شرکت‌کننده \lr{VAF} بالاتری از مدل پایه دارد (بخش~\ref{sec:v2-results}). دوم، صداقت علمی: پرسش «آیا \lr{EEG} چیزی بیش از سینماتیک دارد؟» با مقایسه‌ی \lr{F} و \lr{K} و \lr{residual test} جواب روشن می‌گیرد و پرسش «آیا \lr{EEG} به‌تنهایی حرکت را رمز می‌کند؟» با \lr{E}. سوم، تفسیرپذیری: هر عدد خروجی را می‌توان از خروجی تا ورودی در همه‌ی لایه‌ها دنبال کرد.

\section{مدل \lr{v2}}
\label{sec:v2}

\subsection{ورودی‌ها، \lr{tokens} و خروجی‌ها}

\textbf{نرخ \lr{token}.} همه‌ی سیگنال‌ها پس از بارگذاری، یک بار برای هر سری با فیلتر ضدتداخل بدون تأخیر فاز از ۵۰۰ به ۵۰ هرتز کاهش می‌یابند (ضریب ۱۰). دو سر سری با بازتاب فرد ادامه داده می‌شوند تا اولین و آخرین \lr{token} به سمت صفر کشیده نشوند. هر \lr{token} ۲۰ میلی‌ثانیه است.

\textbf{سینماتیک} $X^{k}\in\mathbb{R}^{T\times24}$: همان ۲۴ ویژگی مدل پایه، استانداردشده با آماره‌های آموزش.

\textbf{تاریخچه} $X^{h}\in\mathbb{R}^{T\times8}$: هر «نگه‌داشتن» بازه‌ای است که نیروی بار \lr{gated} دست‌کم \lr{۰٫۵} ثانیه از ۱ نیوتن بیشتر بماند. یک نگه‌داشتن فقط پس از پایانش برای مدل شناخته می‌شود، پس هر ویژگی در زمان $t$ فقط به گذشته بستگی دارد. هشت ویژگی: پرچم «تاریخچه وجود دارد»، شمار نگه‌داشتن‌های تا اینجا، زمان از شروع سری، زمان از پایان آخرین نگه‌داشتن، میانگین نیروی بار و گرفتن در نگه‌داشتن قبلی، و میانگین جاری نیروی بار و گرفتن در همه‌ی نگه‌داشتن‌های قبلی.

\textbf{\lr{EEG}} $X^{e}\in\mathbb{R}^{T\times32\times4}$: لگاریتم توان در چهار باند تتا (۴ تا ۸ هرتز)، آلفا (۸ تا ۱۳)، بتا (۱۳ تا ۳۰) و گامای پایین (۳۰ تا ۴۰) برای هر الکترود. توان لحظه‌ای با \lr{low-pass} ۵ هرتز هموار، استاندارد و در $\pm5$ بریده می‌شود.

\textbf{هدف در حوزه‌ی مدل.} برای هر عضله $s_c$ صدک ۹۵ پوش آموزش است. مدل در حوزه‌ی جذر کار می‌کند:
\begin{equation}
u=\sqrt{y/s_c},\qquad \hat y=s_c\,\hat u^{2}.
\end{equation}
جذر، قله‌ها را فشرده و سطوح پایین را باز می‌کند؛ به همین دلیل دیگر به وزن‌دهی قله و ماسک استراحت نیازی نیست.

\textbf{خروجی‌ها.} مدل برای هر \lr{token} و هر عضله سه خروجی و سه جزء دقیق برمی‌گرداند:
\begin{equation}
K=K_{base}\odot g(X^{h}),\qquad E=\mathrm{E\text{-}stream}(X^{e}),\qquad F=\mathrm{relu}\big(K+F_{eeg}\big).
\end{equation}

\subsection{بلوک یکسان توجه}

همه‌ی لایه‌های توجه \lr{v2}، چه روی زمان، چه روی الکترودها و چه روی عضله‌ها، یک شکل دارند: بلوک باقی‌مانده با \lr{pre-norm}. برای ورودی $x$ و ماتریس توجه سر $h$:
\begin{equation}
A^{h}=\operatorname{softmax}\!\Big(\frac{Q^{h}K^{h\top}}{\sqrt{d_h}}+M^{h}\Big),\qquad
x\leftarrow x+W_o\big[A^{1}V^{1}\,\|\,\dots\,\|\,A^{H}V^{H}\big],\qquad x\leftarrow x+\mathrm{FFN}(\mathrm{LN}(x)),
\end{equation}
که $Q,K,V$ از $\mathrm{LN}(x)$ ساخته می‌شوند. در توجه زمانی $M=0$ است؛ در لایه‌های گرافی $M^{h}=B^{h}+\lambda P$ که $B^{h}$ \lr{bias} آموختنی هر یال در هر سر، $P$ یک ماتریس پیش‌فرض ثابت و $\lambda$ یک ضریب آموختنی است. سطر $i$ ماتریس $A$ گیرنده و ستون $j$ فرستنده است. \lr{pooling} الکترودها و \lr{cross-attention} \lr{fusion} \lr{residual connection} ندارند.

\paragraph{قاعده‌ی «یک مخلوط‌کننده».} اطلاعات فقط از راه این ماژول توجه و ماژول \lr{attention pooling} بین موقعیت‌ها (زمان، الکترود، عضله، جریان) جابه‌جا می‌شود. تنها استثنا کانولوشن ۵ضربه‌ای \lr{stem} سینماتیک است که ردپایش با ماتریس $S$ (سطرها با جمع ۱، وزن هر جابه‌جایی برابر انرژی هسته در آن جابه‌جایی) صادر می‌شود. همه‌ی لایه‌های دیگر روی یک موقعیت در یک زمان کار می‌کنند. ثبت‌کننده‌ای بیرون از مدل (\lr{with recording(model) as tape}) توجه هر سر را پس از \lr{softmax} و پیش از \lr{dropout} برمی‌دارد و در صورت نیاز گراف گرادیان را نگه می‌دارد. ثبت، خروجی را تغییر نمی‌دهد و هیچ \lr{forward} آرگومان تفسیری ندارد. فهرست لایه‌های ثبت‌شده (۱۳ ماژول) از \lr{model.attention\_layers()} می‌آید.

\subsection{جریان سینماتیک}

\begin{enumerate}
\item \textbf{\lr{stem}:} کانولوشن یک‌بعدی با هسته‌ی ۵ روی $[X^{k}\,\|\,X^{h}]$ (۳۲ ویژگی) به ۶۴ کانال، به‌علاوه‌ی \lr{sinusoidal positional encoding}.
\item \textbf{\lr{temporal encoder}:} ۳ بلوک توجه کامل روی ۴۰۰ \lr{tokens}، ۴ سر، \lr{feed-forward} ۲۵۶، \lr{dropout} برابر \lr{۰٫۲}.
\item \textbf{سر عضله:} تصویر هر \lr{token} به ۵ گره با عرض ۳۲، دو بلوک توجه بین گره‌ها که پیش‌فرض $P$ آن‌ها از هم‌فعالی عضلات در \lr{EMG} آموزش ساخته می‌شود، و یک رمزگشای خطی برای هر عضله با خروجی \lr{softplus} (نامنفی): $K_{base}$.
\item \textbf{\lr{history gain}:} شبکه‌ی کوچک $8\to16\to5$ روی $X^{h}$: $g=2^{\tanh(\cdot)}\in[0.5,2]$، با مقداردهی اولیه‌ای که $g=1$ می‌دهد. $K=K_{base}\odot g$.
\end{enumerate}

\subsection{جریان \lr{EEG}}

\begin{enumerate}
\item \textbf{\lr{embedding} الکترود:} یک \lr{MLP} مشترک چهار \lr{band power} هر الکترود را به ۳۲ بعد می‌برد و یک بردار هویت آموختنی برای هر الکترود اضافه می‌شود. در آموزش هر الکترود با احتمال \lr{۰٫۲} کنار گذاشته می‌شود.
\item \textbf{\lr{electrode graph}:} دو بلوک توجه بین ۳۲ الکترود، برای هر \lr{token} جداگانه، با پیش‌فرض $P_{ij}=-d_{ij}/\operatorname{median}(d)$ که $d_{ij}$ فاصله‌ی دو الکترود در مختصات استاندارد ۱۰-۰۵ است؛ الکترودهای نزدیک به هم \lr{bias} بزرگ‌تری دارند.
\item \textbf{\lr{attention pooling}:} یک پرسش آموختنی روی ۳۲ الکترود توجه می‌کند و یک بردار برای هر \lr{token} می‌سازد: $p_t=\sum_e \pi_{t,e}\,n_{t,e}$.
\item \textbf{\lr{temporal encoder}:} ۲ بلوک توجه روی ۴۰۰ \lr{tokens}.
\item \textbf{سر عضله:} مانند جریان سینماتیک با وزن‌های جدا، خروجی $E$.
\end{enumerate}

\subsection{\lr{fusion}}

\lr{tokens} سینماتیک (با \lr{stop-gradient}) به‌عنوان پرسش روی \lr{tokens} \lr{EEG} توجه می‌کنند و یک لایه‌ی خطی برای هر عضله یک اصلاح می‌دهد:
\begin{equation}
F_{eeg}=\gamma\odot W\,\mathrm{CrossAttn}\big(\mathrm{sg}[Z^{k}],\,Z^{e}\big),\qquad F=\mathrm{relu}\big(\mathrm{sg}[K]+F_{eeg}\big),
\end{equation}
که $\mathrm{sg}$ \lr{stop-gradient} و $\gamma\in\mathbb{R}^{5}$ \lr{gate} هر عضله است که از صفر شروع می‌شود و \lr{weight decay} ندارد. اگر \lr{EEG} چیزی به \lr{K} اضافه نکند، $\gamma$ نزدیک صفر می‌ماند و $F=K$. تعداد کل پارامترها ۳۳۷٬۷۱۲ است.

\subsection{\lr{loss function}}

هر سه خروجی هزینه‌ی یکسان چهارجزئی در حوزه‌ی جذر دارند و هزینه‌ی کل $L=L_K+L_E+L_F$ است:
\begin{equation}
L_o=\underbrace{\mathrm{Huber}_{0.5}(\hat u_o,u)}_{\text{main},\,1.0}
+0.3\underbrace{\big(1-\mathrm{CCC}(\hat u_o,u)\big)}_{\text{ccc}}
+0.5\underbrace{\frac{1}{|\mathcal{H}|}\sum_{k\in\mathcal{H}}\Big|\overline{\hat u_o}_{k}-\overline{u}_{k}\Big|}_{\text{lift}}
+0.1\underbrace{\operatorname{mean}\big|\Delta\hat u_o-\Delta u\big|}_{\text{diff}},
\end{equation}
که $\mathcal{H}$ نگه‌داشتن‌های درون پنجره (با دست‌کم ۵ \lr{tokens}) و $\overline{\cdot}_k$ میانگین روی نگه‌داشتن $k$ است. جزء سطح نگه‌داشتن، خطای سطح فعالیت هر بلندکردن را مستقیم می‌سنجد؛ همان خطایی که در مدل پایه به وزن جسم وابسته بود. عدم‌تقارن، وزن‌دهی قله و ماسک استراحت عمداً حذف شدند.

\subsection{آموزش}

هر جریان \lr{optimizer} خود را دارد: یک \lr{AdamW} برای پارامترهای \lr{kin.*} و یکی برای \lr{eeg.*} و \lr{fusion.*} (\lr{gate} بدون \lr{weight decay})، با نرخ $10^{-3}$، \lr{weight decay} \lr{۰٫۰۱}، ۵ دوره \lr{warm-up} و \lr{cosine decay}، \lr{gradient clipping} \lr{۱٫۰} برای هر جریان جداگانه، \lr{batch size} ۳۲ و حداکثر ۱۵۰ دوره. پنجره‌های آموزش ۸ ثانیه با گام ۲ ثانیه‌اند و آغاز آن‌ها در هر دوره به‌طور تصادفی جابه‌جا می‌شود. پیش از آموزش یک آزمون بودجه چند گام را با و بدون \lr{mixed precision} اندازه می‌گیرد و سریع‌تر را انتخاب می‌کند؛ روی \lr{GTX 1660 Ti} \lr{mixed precision} حدود سه برابر کندتر بود و دقت کامل انتخاب شد.

\paragraph{\lr{freezing} هر جریان.} خطای \lr{validation} هر جریان با میانگین سه‌دوره‌ای هموار می‌شود: $L_K$ برای سینماتیک و $L_E+L_F$ برای \lr{EEG} و \lr{fusion}. شکل~\ref{fig:freeze} قاعده را نشان می‌دهد. جریانی که ۳۰ دوره بهتر نشود به وزن‌های بهترین دوره‌اش برمی‌گردد و دیگر به‌روز نمی‌شود (بدون هزینه، بدون گام \lr{optimizer} و در حالت ارزیابی)، و جریان دیگر ادامه می‌دهد. چون $F$ روی $K$ ساخته می‌شود، وقتی $K$ منجمد شد شمارش صبر \lr{EEG} و \lr{fusion} از نو آغاز می‌شود تا بهترین دوره‌ی آن‌ها با همان $K$ ذخیره‌شده سنجیده شده باشد. آموزش وقتی تمام می‌شود که هر دو جریان منجمد شده باشند یا به ۱۵۰ دوره برسد.
