\section{نتایج ارزیابی \lr{v2}}
\label{sec:v2-results}

همه‌ی اعداد این بخش روی همان سری‌های آزمون مدل پایه و با نرخ ۵۰۰ هرتز حساب شده‌اند. آموزش هر شرکت‌کننده ۱۷ تا ۳۱ دقیقه روی \lr{GTX 1660 Ti} طول کشید (جدول~\ref{tab:v2-train}).

\subsection{دقت \lr{K}، \lr{E} و \lr{F}}

جدول~\ref{tab:v2-subj} خلاصه‌ی اصلی است. \lr{K} با میانگین \lr{VAF} برابر \lr{۷۵٫۸} درصد و \lr{r} برابر \lr{۰٫۸۷۲}، در هر ۱۲ شرکت‌کننده از مدل پایه (\lr{۷۲٫۲} درصد و \lr{۰٫۸۵۱}) بهتر است و از مدل فقط‌سینماتیکِ \lr{residual experiment} (\lr{۷۵٫۵} درصد و \lr{۰٫۸۶۹}) هم کمی جلوتر است؛ بنابراین معیار پذیرش اول طرح \lr{v2} برآورده شد. بهبود \lr{VAF} نسبت به مدل پایه از \lr{۰٫۳} واحد (\lr{P5}) تا \lr{۵٫۵} واحد (\lr{P4}) است و میانه‌ی آن حدود ۴ واحد.

\input{tables/v2_subjects}

\lr{E} که فقط \lr{EEG} می‌بیند به \lr{r} برابر \lr{۰٫۶۴۵} و \lr{VAF} برابر \lr{۳۹٫۱} درصد رسید. این عدد از خروجی فقط‌\lr{EEG} آزمایش‌های قبلی (\lr{r} حدود \lr{۰٫۴۷}) بالاتر است و در هر ۱۲ شرکت‌کننده از \lr{null test} جابه‌جایی بلوک‌های \lr{EEG} گذشت (جدول~\ref{tab:v2-verdict}). یعنی \lr{EEG} به‌تنهایی حالت حرکت را به‌خوبی رمز می‌کند.

\lr{F} عملاً با \lr{K} برابر است: میانگین \lr{VAF} برابر \lr{۷۵٫۹} در برابر \lr{۷۵٫۸}. بزرگ‌ترین قدرمطلق \lr{gate} در هر شرکت‌کننده بین \lr{۰٫۰۱} و \lr{۰٫۰۴} ماند و اصلاح \lr{EEG} حدود \lr{۰٫۲} تا ۲ درصد اندازه‌ی \lr{K} است (جدول~\ref{tab:interp-parts}). فقط در یک شرکت‌کننده (\lr{P4}) بهره‌ی \lr{F} نسبت به \lr{K} در برابر \lr{null test} معنادار شد که با یک مورد از ۱۲ در حد انتظار شانس است. بخش~\ref{sec:residual} نشان می‌دهد این نتیجه خطای \lr{fusion} نیست.

\input{tables/v2_verdict}

\paragraph{قاعده‌ی تفسیرپذیری \lr{F}.} دفترچه‌ی آموزش بررسی می‌کند که آیا \lr{fusion} ذخیره‌شده با همان \lr{K} ذخیره‌شده آموزش دیده است یا نه. با قاعده‌ی \lr{freezing}، بهترین دوره‌ی \lr{EEG} و \lr{fusion} همیشه بعد از \lr{freezing} \lr{K} است (جدول~\ref{tab:v2-train})، پس این شرط در هر ۱۲ نفر برقرار است. جدول حکم‌های ذخیره‌شده‌ی این گزارش پیش از اصلاح این قاعده ساخته شده و برای ۹ نفر برچسب «غیرقابل‌تفسیر» دارد؛ اعداد آن تغییری نمی‌کنند و با اجرای دوباره‌ی سلول ارزیابی، برچسب‌ها به «\lr{EEG} چیزی به سینماتیک اضافه نمی‌کند» تبدیل می‌شوند (به‌جز \lr{P4}).

\subsection{هر عضله و سقف داده}

جدول~\ref{tab:v2-musc} نشان می‌دهد بهبود \lr{K} در همه‌ی عضله‌ها هست. ستون سقف، \lr{VAF} خودِ هدف پس از \lr{low-pass} ۲ هرتز است؛ یعنی بهترین عددی که یک پیش‌بین هموار می‌تواند بگیرد. \lr{FD} با سقف ۷۴ درصد و \lr{modulation depth} کم (نسبت صدک ۸۰ به ۲۰ حدود ۲) محدودترین عضله است؛ فاصله‌ی \lr{K} تا سقف برای \lr{AD}، \lr{BR} و \lr{ED} حدود ۸ تا ۱۱ واحد است.

\subsection{دامنه در هر وزن}

جدول~\ref{tab:v2-load} \lr{height ratio} در مرحله‌ی نگه‌داشتن را برای سه وزن جسم و همبستگی سطح فعالیت بین بلندکردن‌ها را مقایسه می‌کند. جزء سطح نگه‌داشتن در هزینه و ویژگی‌های تاریخچه برای همین خطا طراحی شده بودند و اثرشان روشن است: مدل پایه وزن‌های سبک و متوسط را \lr{۴٫۵} و \lr{۵٫۵} درصد بیش‌برآورد می‌کرد، در حالی که \lr{height ratio} \lr{K} در هر سه وزن بین \lr{۱٫۰۰۴} و \lr{۱٫۰۰۹} است (معیار پذیرش دوم: بین \lr{۰٫۹۵} و \lr{۱٫۰۵}). همبستگی سطح فعالیت بین بلندکردن‌ها در یک وزن ثابت هم در هر سه وزن بالا رفته است (برای نمونه از \lr{۰٫۳۹} به \lr{۰٫۵۳} در وزن سبک).

\subsection{\lr{triptych figures}}

برای هر سری آزمون، کل سری به بازه‌های ۱۰ ثانیه‌ای پشت سر هم بریده و برای هر بازه یک شکل ساخته شد. در ادامه برای شرکت‌کننده‌های ۱ تا ۴ و هر سری آزمون یک شکل آمده است (ثانیه‌ی ۳۰ تا ۴۰ هر سری). سه پنل هر شکل:
\begin{itemize}
\item \textbf{پنل ۱:} پوش واقعی (سیاه) در برابر \lr{K} (نارنجی)، \lr{E} (آبی) و \lr{F} (سبز خط‌چین)، با واحد اصلی. خطوط عمودی رخدادهای آزمایش‌اند (روشن‌شدن چراغ، اولین تماس، جداشدن از سطح، گذاشتن و رها کردن) و ناحیه‌ی زرد نگه‌داشتن تشخیص‌داده‌شده است. ستون راست \lr{r} همین بازه است.
\item \textbf{پنل ۲:} اجزای دقیق در حوزه‌ی مدل: $K_{base}$ (خاکستری)، $K=K_{base}\cdot g$ (نارنجی) و $F_{eeg}$ (بنفش). ستون راست بازه‌ی \lr{history gain} و نسبت $|F_{eeg}|/|K|$ را می‌دهد.
\item \textbf{پنل ۳:} \lr{attention rollout} \lr{muscle graph} جریان سینماتیک، به‌صورت تغییر نسبت به میانگین استراحت همان سری. هر بلوک یک عضله‌ی گیرنده و هر سطر یک عضله‌ی فرستنده است.
\end{itemize}
در همه‌ی شکل‌ها \lr{F} روی \lr{K} افتاده و $F_{eeg}$ نزدیک صفر است؛ \lr{E} زمان‌بندی کلی بلندکردن را دنبال می‌کند اما جهش‌های کوتاه را ندارد.
