\section{جمع‌بندی نهایی}
\label{sec:conclusion}

این گزارش با یک پرسش ساده شروع شد: آیا \lr{EEG} سطح سر چیزی درباره‌ی فعالیت عضله‌های دست می‌داند که از حرکت خود دست نتوان فهمید؟ این فصل اول آنچه هر فصل به آن رسید را مرور می‌کند، بعد این نتیجه‌ها را کنار هم می‌گذارد و به پرسش اصلی جواب می‌دهد، و در پایان محدودیت‌ها و قدم‌های بعدی را می‌آورد.

\subsection{آنچه هر فصل نشان داد}

\paragraph{داده و پیش‌پردازش (فصل‌های ۱ و ۲).} داده‌ی \lr{WAY-EEG-GAL} از ۱۲ نفر است که جسمی را بارها با دست راست گرفته و بلند کرده‌اند. هدف پیش‌بینی، پوش \lr{EMG} پنج عضله است. پیش‌پردازش \lr{v2} سه سیگنال را هم‌زمان و هم‌نرخ می‌کند: \lr{EEG} به توان چهار باند در ۳۲ الکترود، سینماتیک و نیرو به ۱۳ ویژگی و سرعتشان، و \lr{EMG} به پوش هموار در حوزه‌ی جذر. یک \lr{gate} نزدیکی دست به جسم نیروهایی را که آزمایشگر بین دو بلند کردن به جسم وارد می‌کند حذف می‌کند.

\paragraph{مدل پایه و کاستی‌هایش (فصل‌های ۳ و ۴).} مدل پایه (\lr{KG-GT}) با یک خروجی به \lr{VAF} \lr{۷۲٫۲} درصد رسید، اما چون \lr{EEG} و سینماتیک درون یک گراف مخلوط می‌شدند، نمی‌شد گفت هر کدام چه سهمی دارند. آزمون‌های حذف ورودی نشان دادند تقریباً همه‌ی دقت از سینماتیک می‌آید.

\paragraph{مدل \lr{v2} و نتایجش (فصل‌های ۵ تا ۷).} \lr{v2} سه خروجی جدا دارد و به همین دلیل می‌تواند سه پرسش را جدا جواب دهد:
\begin{itemize}
\item \textbf{سینماتیک:} \lr{K} با \lr{VAF} \lr{۷۵٫۸} درصد و $r$ \lr{۰٫۸۷۲} در هر ۱۲ نفر از مدل پایه بهتر است. پنجره‌ی ۸ ثانیه‌ای، ویژگی‌های تاریخچه، حوزه‌ی جذر و جزء \lr{lift} در \lr{loss function} پشت این بهبودند.
\item \textbf{\lr{EEG} به‌تنهایی:} \lr{E} با $r$ \lr{۰٫۶۴۵} در هر ۱۲ نفر از \lr{null test} گذشت. پس \lr{EEG} به‌تنهایی اطلاعات واقعی درباره‌ی فعالیت عضله دارد.
\item \textbf{\lr{EEG} به‌علاوه‌ی سینماتیک:} \lr{F} با \lr{K} برابر است و \lr{gate} نزدیک صفر می‌ماند.
\end{itemize}

\paragraph{تفسیرپذیری (فصل‌های ۸ و ۹).} قاعده‌ی «یک مخلوط‌کننده» و \lr{recorder} بیرونی باعث شدند \lr{rollout}، \lr{attention flow}، \lr{gradient-weighted rollout} و نمای هر لایه برای هر خروجی و در همه‌ی لایه‌ها کار کنند. خود این روش‌ها هم سنجیده شدند. \lr{gradient-weighted rollout} قابل‌اعتمادترین بود: در \lr{deletion check} بهترین است و کمترین شباهت را با مدل به‌هم‌ریخته دارد. \lr{attention flow} در این معماری به سقف می‌رسد و فقط در مقایسه با خط پایه‌ی توجه یکنواخت معنا دارد. \lr{K} در همه‌ی افراد به حرکت پیش از هر رخداد نگاه می‌کند.

\paragraph{یکسانی بین افراد و مقایسه با پژوهش‌ها (فصل‌های ۱۰ و ۱۱).} نقشه‌ی الکترود یک نفر پرنویز است (\lr{pairwise $\rho$} حدود \lr{۰٫۲})، ولی نقشه‌ی گروه تکرارپذیر است (\lr{split-half} حدود \lr{۰٫۷}) و دو کانون معنادار روی \lr{C3} و \lr{CP1} دارد. \lr{E} باند بتا را ترجیح می‌دهد و قله‌ی توجهش حدود \lr{۰٫۵} تا \lr{۰٫۸} ثانیه پیش از رخدادهاست. هر دو با پژوهش‌های \lr{ERD} می‌خوانند. اما \lr{E} عضله‌ی شانه را بهتر از عضله‌های انگشت پیش‌بینی می‌کند و برای هر عضله نقشه‌ی خاصی ندارد؛ این با پژوهش‌های \lr{CMC} نمی‌خواند.

\paragraph{\lr{residual test} (فصل ۱۲).} \lr{EEG} اطلاعات کوچک ولی واقعی درباره‌ی خطای \lr{K} دارد: $r$ حدود \lr{۰٫۰۷}، معنادار در ۱۰ نفر از ۱۲، که کمتر از نیم واحد \lr{VAF} اضافه می‌کند. \lr{fusion} همین اطلاعات را پیدا کرده بود. پس $F\approx K$ نتیجه‌ی درست داده است، نه نقص مدل.

\paragraph{سه آزمایش برای مقاله (فصل ۱۳).}
\begin{itemize}
\item یک مدل خطی که فقط زمان رخدادهای تکلیف را می‌داند در هر ۱۲ نفر از \lr{E} بهتر است ($r$ \lr{۰٫۸۱} در برابر \lr{۰٫۶۴}) و ترتیب عضله‌ها را هم دقیقاً مثل \lr{E} دارد. فقط دانستن زمان روشن شدن \lr{LED} تا حدود ۹۰ درصد دقت \lr{E} را می‌دهد.
\item \lr{ERD} در هر ۱۲ نفر الگوی کلاسیک دارد و سه الکترود اولش (\lr{C3}، \lr{CP1} و \lr{C4}) همان سه الکترود اول \lr{E} است. نقشه‌ی \lr{E} در سطح گروه به‌طور معنادار با \lr{ERD} بتا و \lr{CMC} هم‌جهت است، ولی همبستگی متوسط است ($\rho$ حدود \lr{۰٫۲}) و \lr{E} از الکترودهای کنار چشم و پشت گردن هم می‌خواند.
\item یک نسخه‌ی علّی از \lr{v2} برای \lr{K} بدون هزینه کار می‌کند. وقتی مدل \lr{EMG} را جلوتر پیش‌بینی می‌کند، سهم \lr{EEG} کمی بیشتر می‌شود: $F-K$ در ۵۰۰ میلی‌ثانیه \lr{+۰٫۵۵} واحد \lr{VAF} (معنادار) و همبستگی \lr{EEG} با خطای \lr{K} پیش از حرکت از \lr{۰٫۰۶} به \lr{۰٫۱۹}. این اطلاعات فقط در رساندن دست به کاهش واقعی خطا تبدیل می‌شود (تا ۵ درصد).
\end{itemize}

\subsection{پاسخ به پرسش اصلی}

اگر همه‌ی نتیجه‌ها کنار هم گذاشته شوند، تصویر یکپارچه‌ای به دست می‌آید. \lr{EEG} در این داده سه لایه اطلاعات درباره‌ی فعالیت عضله دارد، و هر لایه کوچک‌تر از قبلی است.

\paragraph{لایه‌ی اول: زمان‌بندی تکلیف.} بیشترِ آنچه \lr{E} می‌داند این است که «الان در کدام مرحله‌ی تکلیف هستیم». سه شاهد مستقل این را نشان می‌دهند. مدل زمان‌بندی در هر ۱۲ نفر از \lr{E} بهتر است. ترتیب عضله‌ها در \lr{E} همان ترتیب مدل زمان‌بندی است، پس دلیلی ندارد که از خود مغز بیاید. نقشه‌های عضله‌ها در \lr{E} هم تقریباً یکسان‌اند. این لایه بزرگ است و \lr{E} را به $r$ حدود \lr{۰٫۶} می‌رساند. اما سینماتیک هم همین اطلاعات را دارد، پس چیزی به \lr{K} اضافه نمی‌کند.

\paragraph{لایه‌ی دوم: منبع قشری بخشی از این زمان‌بندی.} \lr{E} زمان‌بندی را دست‌کم تا حدی از جای درست می‌خواند. سه الکترود اول \lr{E} همان سه الکترود اول \lr{ERD} بتا هستند، \lr{E} باند بتا را ترجیح می‌دهد، و قله‌ی توجهش در همان بازه‌ای است که \lr{ERD} پیش از حرکت شروع می‌شود. همبستگی نقشه‌ها در سطح گروه معنادار است. اما بخش دیگری از اطلاعات \lr{E} از الکترودهای کنار چشم، گیجگاه و پشت گردن می‌آید، و مدل \lr{cue} نشان داد که پاسخ به یک محرک بینایی به‌تنهایی برای بیشتر دقت \lr{E} کافی است. با داده‌ی فعلی نمی‌توان گفت چه سهمی از \lr{E} قشری و چه سهمی \lr{artifact} یا پاسخ بینایی است.

\paragraph{لایه‌ی سوم: اطلاعاتی که سینماتیک ندارد.} این لایه بسیار کوچک است. در پیش‌بینی هم‌زمان، \lr{EEG} کمتر از نیم واحد \lr{VAF} اضافه می‌کند. وقتی مدل باید آینده را پیش‌بینی کند، سهم \lr{EEG} کمی بیشتر می‌شود و دقیقاً در جای مورد انتظار: پیش از حرکت و در رساندن دست، نه در نگه‌داشتن. این قوی‌ترین نتیجه‌ی مثبت گزارش برای \lr{EEG} است. با این حال، حتی این لایه می‌تواند از زمان‌بندی بیاید: سینماتیک پیش از حرکت نمی‌داند حرکت کی شروع می‌شود، ولی \lr{EEG} ممکن است آن را از پاسخ به چراغ \lr{LED} بداند. اینکه این اطلاعات در \lr{band power} خام با افق کم می‌شود، این توضیح را محتمل‌تر می‌کند.

\paragraph{در یک جمله.} \lr{EEG} سطح سر در این داده بیشتر می‌داند حرکت \emph{کی} اتفاق می‌افتد و تقریباً چیزی درباره‌ی اینکه هر عضله \emph{چقدر} فعال است فراتر از سینماتیک نمی‌داند. تنها جایی که سهم مستقل و قابل‌اندازه‌گیری دارد، پیش‌بینی چند صد میلی‌ثانیه‌ی بعد در آغاز حرکت است.

\subsection{ادعای قابل دفاع برای مقاله}

نتایج این گزارش از این ادعا پشتیبانی می‌کنند: «در تکلیف گرفتن و بلند کردن، دقت بالای مدل‌هایی که \lr{EMG} را از \lr{EEG} پیش‌بینی می‌کنند می‌توان تقریباً به‌طور کامل با زمان‌بندی تکلیف توضیح داد؛ \lr{EEG} فقط در پیش‌بینی آینده‌ی نزدیک و در آغاز حرکت اطلاعاتی فراتر از سینماتیک دارد، و منبع آن هنوز روشن نیست.» این ادعا سه ویژگی خوب دارد. با چهار روش مستقل آزموده شده است (\lr{residual test}، خط پایه‌ی زمان‌بندی، نقشه‌های مرجع و افق پیش‌بینی). در هر ۱۲ نفر با آزمون گروهی سنجیده شده است. و مستقیم به یک مشکل روش‌شناختی در این حوزه اشاره می‌کند: دقت یک مدل \lr{EEG} بدون مقایسه با خط پایه‌ی زمان‌بندی معنای روشنی ندارد.

برای قوی‌تر شدن این ادعا، دو آزمایش از همه مهم‌ترند. هر دو در بخش قدم‌های بعدی آمده‌اند: کنترل \lr{artifact} با زیرمجموعه‌ی الکترودها، و خط پایه‌ی علّی زمان‌بندی در کنار مدل علّی.

\subsection{محدودیت‌ها}
\begin{itemize}
\item \lr{rollout} و \lr{attention flow} فقط وزن‌های توجه را می‌بینند، و وزن \lr{۰٫۵} \lr{residual connection} یک قرارداد است. روش‌های گرادیانی فقط رفتار مدل را در نزدیکی ورودی فعلی و به‌صورت خطی می‌سنجند.
\item \lr{plain rollout} \lr{K} برای مدل به‌هم‌ریخته هم شکل مشابهی دارد (همبستگی \lr{۰٫۴۷})؛ پس شکل آن بیشتر از ساختار مدل می‌آید تا از یادگیری.
\item پنجره‌های رخدادهای نزدیک به هم هم‌پوشانی دارند، پس $p$ مقایسه‌ی رخداد با استراحت تقریبی است.
\item هر مدل برای یک شرکت‌کننده آموزش دیده است؛ پیش‌بینی برای فرد تازه آزموده نشده است.
\item مدل اصلی برون‌خط است. نسخه‌ی علّی یک نشت کوچک دارد: \lr{gate} نیرو تا حدود ۵۰ میلی‌ثانیه آینده را می‌بیند.
\item خط پایه‌ی زمان‌بندی زمان رخدادها را از کل ضبط می‌خواند؛ سقف است، نه پیش‌بینی‌کننده‌ی واقعی.
\item \lr{CMC} در این داده ضعیف است و برای بیشتر عضله‌ها کانونش روی پس‌سر است؛ مرجع آلوده‌ای است.
\item هر افق فقط یک بار برای هر نفر آموزش دید؛ تغییرپذیری بین اجراها (مثلاً افت \lr{P4} و \lr{P5} در یک ثانیه) اندازه گرفته نشده است.
\item سقف داده (به‌ویژه برای \lr{FD}) هر بهبود بیشتری را محدود می‌کند.
\end{itemize}

\subsection{قدم‌های بعدی}
\begin{enumerate}
\item \textbf{کنترل \lr{artifact} با زیرمجموعه‌ی الکترودها.} \lr{E} یک بار فقط با الکترودهای مرکزی (\lr{C3}، \lr{C4}، \lr{Cz} و \lr{CP}ها) و یک بار فقط با الکترودهای کناری، پیشانی و پس‌سری آموزش ببیند. اگر نسخه‌ی مرکزی دقت \lr{E} و رشد اطلاعات پیش از حرکت را نگه دارد، منبع قشری تأیید می‌شود.
\item \textbf{خط پایه‌ی علّی زمان‌بندی.} مدل علّی \lr{K} به‌علاوه‌ی فقط زمان رخدادهای \emph{گذشته} (به‌ویژه روشن شدن \lr{LED}) در همان افق‌ها. اگر \lr{EEG} از این مدل هم بهتر باشد، سهمش فراتر از دانستن زمان چراغ است.
\item \textbf{\lr{ridge} جدا برای هر مرحله.} برازش جدا برای مرحله‌ی پیش از حرکت، تا همبستگی بالای آن مرحله به مهارت تبدیل شود یا معلوم شود که نمی‌شود.
\item \textbf{کنترل حرکت چشم.} حذف یا رگرسیون کانال‌های نزدیک چشم پیش از آموزش، و تکرار آزمایش افق.
\item \textbf{مدل بین افراد.} آموزش روی ۱۱ نفر و آزمون روی نفر دوازدهم، تا معلوم شود الگوی قشری بین افراد مشترک است یا نه.
\end{enumerate}

\paragraph{بازتولید.} کد در شاخه‌ی \lr{KG-GT-v2} است (بسته‌ی \lr{v2/} و دفترچه‌ی \lr{main\_v2.ipynb}). جدول‌ها و شکل‌های فصل یکسانی بین شرکت‌کننده‌ها را اسکریپت \lr{v2-report/analysis/consistency.py} از نتایج ذخیره‌شده می‌سازد. آزمایش‌های فصل \ref{sec:controls} در دفترچه‌ی \lr{eeg\_controls.ipynb} و بسته‌ی \lr{eeg\_controls/} اجرا می‌شوند و جدول‌ها و شکل‌هایشان را \lr{v2-report/analysis/eeg\_controls\_results.py} می‌سازد. هیچ‌کدام از این دو اسکریپت مدلی را اجرا نمی‌کنند.
