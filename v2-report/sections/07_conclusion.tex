\section{جمع‌بندی و محدودیت‌ها}

\paragraph{آنچه به دست آمد.} مدل پایه سه پرسش را در یک خروجی با هم قاطی می‌کرد. \lr{v2} با سه خروجی جدا به هر کدام جداگانه جواب داد.
\begin{enumerate}
\item \textbf{سینماتیک چقدر خوب پیش‌بینی می‌کند؟} \lr{K} با \lr{VAF} \lr{۷۵٫۸} درصد و $r$ \lr{۰٫۸۷۲} در هر ۱۲ شرکت‌کننده از مدل پایه بهتر است و دامنه را در هر سه وزن جسم با خطای کمتر از یک درصد پیش‌بینی می‌کند. چهار تغییر پشت این بهبودند: پنجره‌ی ۸ ثانیه‌ای، ویژگی‌های تاریخچه، حوزه‌ی جذر و جزء \lr{lift} در \lr{loss function}.
\item \textbf{آیا \lr{EEG} به‌تنهایی چیزی از حرکت می‌داند؟} بله. \lr{E} با $r$ \lr{۰٫۶۴۵} در هر ۱۲ نفر از \lr{null test} گذشت. اما آنچه می‌خواند بیشتر مرحله‌ی کلی حرکت است (رسیدن، گرفتن، بلند کردن، استراحت)، نه فرمان خاص هر عضله.
\item \textbf{آیا \lr{EEG} چیزی بیش از سینماتیک دارد؟} تقریباً نه. \lr{F} با \lr{K} برابر است. \lr{residual test} سیگنالی واقعی ولی بسیار کوچک یافت ($r$ حدود \lr{۰٫۰۷}، کمتر از نیم واحد \lr{VAF})، و \lr{fusion} هم همین را یافته بود.
\end{enumerate}

\paragraph{تفسیرپذیری.} قاعده‌ی «یک مخلوط‌کننده» و \lr{recorder} بیرونی باعث شدند \lr{rollout}، \lr{attention flow}، \lr{gradient-weighted rollout} و نمای هر لایه برای هر خروجی و در همه‌ی لایه‌ها بدون حالت خاص کار کنند. آزمون‌های روی خودِ روش‌ها نشان دادند \lr{gradient-weighted rollout} قابل‌اعتمادترین است: در \lr{deletion check} بهترین است و کمترین شباهت را با مدل به‌هم‌ریخته دارد. \lr{attention flow} در این معماری به سقف می‌رسد و فقط در مقایسه با خط پایه‌ی توجه یکنواخت معنا دارد.

در سطح گروه، نقشه‌ی الکترود \lr{E} کانون تکرارپذیری روی \lr{C3} و \lr{CP1} دارد، بالای قشر حسی-حرکتی نیمکره‌ی مقابل دست، و باند بتا را ترجیح می‌دهد؛ هر دو با پژوهش‌های پیشین می‌خوانند. نقشه‌ی هر فرد جداگانه پرنویز است. سهم الکترودهای پیشانی و پس‌سری هشدار می‌دهد که بخشی از اطلاعات \lr{E} ممکن است از \lr{artifact} باشد.

\paragraph{محدودیت‌ها.}
\begin{itemize}
\item \lr{rollout} و \lr{attention flow} فقط وزن‌های توجه را می‌بینند، و وزن \lr{۰٫۵} \lr{residual connection} یک قرارداد است. روش‌های گرادیانی فقط رفتار مدل را در نزدیکی ورودی فعلی و به‌صورت خطی می‌سنجند.
\item \lr{plain rollout} \lr{K} برای مدل به‌هم‌ریخته هم شکل مشابهی دارد (همبستگی \lr{۰٫۴۷})؛ پس شکل آن بیشتر از ساختار مدل می‌آید تا از یادگیری.
\item پنجره‌های رخدادهای نزدیک به هم هم‌پوشانی دارند، پس $p$ مقایسه‌ی رخداد با استراحت تقریبی است.
\item هر مدل برای یک شرکت‌کننده آموزش دیده است؛ پیش‌بینی برای فرد تازه آزموده نشده است.
\item پیش‌بینی برون‌خط است: هر \lr{token} از کل پنجره‌ی ۸ ثانیه‌ای، از جمله لحظه‌های بعد از خودش، استفاده می‌کند.
\item سقف داده (به‌ویژه برای \lr{FD}) هر بهبود بیشتری را محدود می‌کند.
\end{itemize}

\paragraph{بازتولید.} کد در شاخه‌ی \lr{KG-GT-v2} است (بسته‌ی \lr{v2/} و دفترچه‌ی \lr{main\_v2.ipynb}). جدول‌ها و شکل‌های فصل یکسانی بین شرکت‌کننده‌ها را اسکریپت \lr{v2-report/analysis/consistency.py} از نتایج ذخیره‌شده می‌سازد و هیچ مدلی را اجرا نمی‌کند.
