\section{مدل \lr{v3}: \lr{EEG} پایه، و آنچه \lr{EEG} فراتر از زمان‌بندی اضافه می‌کند}
\label{sec:v3}

\subsection{چرا \lr{v3}}

\lr{v2} نشان داد \lr{E} از آزمون \lr{null} می‌گذرد ولی \lr{F} با \lr{K} برابر می‌ماند. دو نکته این نتیجه را ناتمام گذاشت.

\paragraph{جهت مدل.} در \lr{v2}، \lr{K} پایه بود و \lr{EEG} فقط یک اصلاح کوچک روی آن. وقتی سینماتیک خودش حدود ۷۶ درصد \lr{VAF} می‌دهد، چنین ساختاری طبیعی است که \lr{EEG} را نادیده بگیرد. هم‌تیمی در بازبینی گفت قرار بود \lr{EEG} پایه باشد و سینماتیک فقط راهنمایی‌اش کند، و از این طراحی نمی‌شود نتیجه گرفت \lr{EEG} اثری ندارد. قاعده‌ی پروژه هم همین شد: سینماتیک فقط در آموزش راهنماست و در زمان آزمون مدل فقط از \lr{EEG} پیش‌بینی می‌کند.

\paragraph{ساعت تکلیف.} مدل خطی \lr{phase} که فقط زمان رخدادها را می‌داند به $r$ برابر \lr{۰٫۸۱۴} می‌رسد و از \lr{E} بهتر است. پس پرسش منصفانه این نیست که \lr{EEG} به‌تنهایی چقدر خوب است. پرسش این است که وقتی حرکت گذشته و زمان رخدادهای گذشته را داریم، \lr{EEG} چیزی اضافه می‌کند یا نه. نسخه‌ی اول \lr{v3} رسیدن به سطح مدل \lr{phase} را موفقیت حساب می‌کرد. این معیار کنار گذاشته شد، چون مدلی که مغز ندارد همین حالا به آن سطح می‌رسد: رسیدن به آن نتیجه‌ی \lr{null} است، نه هدف.

\subsection{مقایسه‌ی اصلی: \lr{B} در برابر \lr{F}}

\paragraph{ساختن \lr{K}، \lr{B} و \lr{F}.} \lr{K} جریان سینماتیک مدل علّی \lr{v2} است که برای هر افق و هر نفر از پیش آموزش دیده بود (\lr{outputs/checkpoints\_v2\_horizon}). \lr{B} یک \lr{ridge} روی خطای \lr{K} اضافه می‌کند که ورودی‌اش زنگوله‌های گاوسی بعد از هر رخداد گذشته است (\lr{LED}، شروع حرکت دست، تماس، بلند شدن جسم، گذاشتن، رها کردن و توقف دست). \lr{F} همان \lr{ridge} است با یک ورودی بیشتر: \lr{token}های \lr{EEG} همان مدل علّی. هر دو \lr{ridge} روی سری \lr{validation} برازش و روی سری‌های آزمون سنجیده می‌شوند، چون \lr{K} روی سری‌های آموزش آموزش دیده است. پس \lr{B} و \lr{F} از یک خانواده‌اند، همان هدف و همان کالیبراسیون را دارند و فقط در \lr{EEG} فرق می‌کنند.

\paragraph{معیارها.} معیار اصلی $\text{\lr{eeg\_skill}} = 1 - \text{\lr{MSE}}(F)/\text{\lr{MSE}}(B)$ است، برای هر نفر، افق و فاز تکلیف (پیش از حرکت، رساندن دست، بار، نگه‌داشتن، رها کردن، استراحت و کل). مقدار ۰ یعنی \lr{EEG} چیزی اضافه نمی‌کند و ۰٫۰۵ یعنی پنج درصد خطای کمتر. $p$ هر نفر از یک \lr{null} می‌آید که فقط سهم \lr{EEG} را در زمان جابه‌جا می‌کند و سهم رخدادها را سر جایش می‌گذارد. آزمون گروهی \lr{Wilcoxon} یک‌طرفه روی ۱۲ نفر است، با تصحیح \lr{Holm} روی افق‌ها (آزمون اصلی) و جداگانه روی فازها. معیارهای فرعی عبارت‌اند از اختلاف \lr{VAF}، \lr{CCC}، خطای مطلق هر شرط و خطای زمان شروع فعالیت (\lr{onset}): لحظه‌ای که پوش بعد از چراغ به ۲۰ درصد فاصله‌ی پایه تا قله می‌رسد.

\paragraph{کنترل‌ها.} \lr{EEG} ورودی \lr{F} از پنج منبع می‌آید:
\begin{itemize}
\item منبع \lr{eeg}: \lr{EEG} واقعی، یعنی خود \lr{F}.
\item منبع \lr{phase\_swap}: \lr{EEG} یک \lr{trial} دیگر با همان وزن و همان سطح، که زمانش طوری کشیده یا فشرده شده که همه‌ی رخدادهای دو \lr{trial} روی هم بیفتند. زمان‌بندی تکلیف حفظ می‌شود و فقط \lr{EEG} مال \lr{trial} دیگری است. اگر سهم \lr{EEG} مخصوص هر \lr{trial} باشد، این منبع باید به‌وضوح بدتر از \lr{eeg} باشد.
\item منبع \lr{swap}: همین، با هم‌تراز کردن فقط چراغ \lr{LED}.
\item منبع \lr{bandpower}: توان باند همه‌ی کانال‌ها، بدون شبکه‌ی آموزش‌دیده.
\item منبع \lr{peripheral}: همین توان، فقط روی کانال‌های قطب پیشانی، گیجگاهی و پس‌سری که بیشتر حرکت چشم و فعالیت عضله‌ی سر و گردن را می‌گیرند. اگر این منبع به اندازه‌ی \lr{bandpower} کمک کند، سهم \lr{EEG} احتمالاً از \lr{artifact} است.
\end{itemize}

\paragraph{تفسیر.} ادعای «\lr{EEG} چیزی فراتر از حرکت و زمان‌بندی دارد» فقط وقتی پذیرفته است که سه چیز با هم درست باشد: \lr{F} پس از \lr{Holm} از \lr{B} بهتر باشد، از \lr{phase\_swap} به‌طور معنادار بهتر باشد و \lr{peripheral} آن را توضیح ندهد. اگر \lr{F} از \lr{B} بهتر باشد ولی یکی از دو کنترل را رد کند، سهم \lr{EEG} به تکلیف قفل است یا از \lr{artifact} می‌آید. ترکیب این نتیجه با پرسش ثانویه، جدول تفسیر پیشنهاد را می‌سازد (جدول~\ref{tab:v3-interp}).

\subsection{آزمایش ثانویه: مدل فقط‌\lr{EEG} با راهنمای سینماتیک}

مدل \lr{student} همان جریان \lr{EEG} مدل \lr{v2} است و ورودی‌اش فقط توان باند \lr{EEG} است. معلم، جریان سینماتیک \lr{v2} همان نفر است که وزن‌هایش ثابت است و هرگز همراه مدل ذخیره نمی‌شود. راهنما دو جمله به \lr{loss} اضافه می‌کند: \lr{align} که \lr{token}های \lr{EEG} را از راه یک لایه‌ی خطی به \lr{token}های معلم نزدیک می‌کند، و \lr{distill} که خروجی را به خروجی \lr{K} نزدیک می‌کند. وزن این کمک برابر \lr{scale} ضرب در یک برنامه‌ی زمانی است: تا \lr{epoch} ۴۰ کامل است، تا \lr{epoch} ۶۰ خطی به صفر می‌رسد و بعد صفر می‌ماند. پس پایان هر آموزش فقط \lr{EEG} به \lr{EMG} است. توقف زودهنگام و انتخاب بهترین \lr{epoch} فقط با \lr{loss} \lr{validation} خود \lr{E} انجام می‌شود. \lr{E0} همان مدل با \lr{scale} صفر است. سه مقدار \lr{scale} (۰٫۰۵، ۰٫۱ و ۰٫۳) و سه \lr{seed} برای هر نفر آموزش دیده‌اند.

\subsection{کنترل‌های دیگر}

\begin{itemize}
\item \textbf{مدل‌های زمان‌بندی علّی.} مدل \lr{causal\_cue} فقط زمان گذشته از آخرین \lr{LED} را می‌داند و مدل \lr{events\_causal} زمان گذشته از آخرین وقوع هر کدام از هشت رخداد را. هر دو فقط گذشته را می‌بینند. مدل‌های \lr{phase} و \lr{cue} در \lr{v2} آینده را هم می‌دیدند و حالا فقط سقف‌اند.
\item \textbf{آزمون انحراف از الگو.} برای هر نفر و عضله، یک الگوی میانگین \lr{trial} فقط از سری‌های آموزش ساخته می‌شود و آزمون می‌پرسد آیا \lr{EEG} انحراف هر \lr{trial} آزمون از این الگو را پیش‌بینی می‌کند یا نه.
\item \textbf{امتیازدهی در دو نرخ.} برای جواب دادن به نگرانی هم‌تیمی درباره‌ی کاهش و افزایش نرخ، مدل‌ها هم در ۵۰۰ هرتز (مثل \lr{v2}) و هم در ۵۰ هرتز (بدون \lr{upsample}) سنجیده می‌شوند.
\item \textbf{بررسی \lr{split} معلم.} پیش از آموزش، کد بررسی می‌کند که معلم روی سری‌های \lr{validation} یا آزمون این \lr{run} آموزش ندیده باشد.
\end{itemize}
