\section{\lr{residual test}: آیا \lr{EEG} چیزی بیش از سینماتیک دارد؟}
\label{sec:residual}

\subsection{پرسش}

\lr{F} برابر \lr{K} ماند و \lr{gate} نزدیک صفر. دو توضیح ممکن است: یا \lr{EEG} واقعاً اطلاعاتی درباره‌ی خطای \lr{K} ندارد، یا دارد و ماژول \lr{fusion} نتوانسته آن را یاد بگیرد. \lr{residual test} این دو را بدون ماژول \lr{fusion} از هم جدا می‌کند: مستقیم می‌پرسد آیا از \lr{EEG} می‌توان باقی‌مانده‌ی \lr{K} را پیش‌بینی کرد.

\subsection{روش}

\textbf{باقی‌مانده.} در حوزه‌ی مدل و با نرخ \lr{token}: $\varepsilon=u-\hat u_K$ که $u=\sqrt{y/s_c}$ هدف و $\hat u_K$ خروجی \lr{K} است.

\textbf{ویژگی‌ها.} دو مجموعه، هر کدام برای هر \lr{token}:
\begin{itemize}
\item \textbf{\lr{tokens}:} خروجی \lr{temporal encoder} جریان \lr{EEG} (۶۴ بعد)، یعنی همان چیزی که ماژول \lr{fusion} می‌خواند؛
\item \textbf{\lr{band power}:} ۱۲۸ ویژگی (۳۲ الکترود × ۴ باند) ورودی مدل، با \lr{moving average} یک‌ثانیه‌ای. این مجموعه به جریان \lr{EEG} آموزش‌دیده وابسته نیست.
\end{itemize}

\textbf{برازش.} \lr{ridge regression} با ورودی استانداردشده و عرض از مبدأ:
\begin{equation}
\hat W=\arg\min_W\ \|\varepsilon-ZW\|^2+\alpha\,N\,\|W\|^2,
\end{equation}
روی \emph{سری‌های \lr{validation}} برازش می‌شود؛ \lr{K} روی این سری‌ها آموزش ندیده است، پس باقی‌مانده‌ی آن‌ها شبیه باقی‌مانده‌ی سری‌های آزمون است. در سری‌های آموزش، \lr{K} داده را تا حدی حفظ کرده و باقی‌مانده کوچک‌تر از واقع است. $\alpha$ از میان $10^{-2},\dots,10^{4}$ با \lr{cross-validation} روی پنج بلوک پیوسته‌ی زمانی انتخاب می‌شود.

\textbf{سنجش روی سری‌های آزمون.} برای هر عضله:
\begin{equation}
r=\operatorname{corr}(\hat\varepsilon,\varepsilon),\qquad
\Delta\mathrm{VAF}=\mathrm{VAF}\big(s_c\,\mathrm{relu}(\hat u_K+\hat\varepsilon)^2,\ y\big)-\mathrm{VAF}\big(s_c\,\hat u_K^{2},\ y\big).
\end{equation}

\textbf{\lr{null distribution}.} همان پیش‌بینی $\hat\varepsilon$ به‌طور دایره‌ای در زمان جابه‌جا می‌شود و دوباره سنجیده می‌شود (۹۹ بار). جابه‌جایی همیشه مضربی از گام پیش‌بینی (۲۰۰ \lr{tokens}، یعنی ۴ ثانیه) و دست‌کم یک پنجره است. دلیل این قید در بخش بعد آمده است. $p$ یک‌طرفه است و هیچ‌گاه کمتر از یک‌تقسیم‌بر‌تعداد جابه‌جایی‌های ممکن گزارش نمی‌شود (۱۵۲ تا ۱۹۰ جابه‌جایی در شرکت‌کننده‌های مختلف). آزمون اصلی میانگین روی پنج عضله است.

\textbf{مقایسه با مدل.} همبستگی اصلاحی که خود مدل ساخته ($F-K$) با باقی‌مانده هم گزارش می‌شود: $r_{model}$.

\subsection{یک دام: موقعیت درون پنجره}

\lr{encoders} زمانی به هر \lr{token} \lr{sinusoidal positional encoding} اضافه می‌کنند، پس \lr{tokens} \lr{EEG} «جای خود در پنجره‌ی ۸ ثانیه‌ای» را می‌دانند. پیش‌بینی پیوسته‌ی \lr{K} هم ممکن است خطای کوچکی با همین دوره داشته باشد (هر ۴ ثانیه یک پنجره‌ی تازه آغاز می‌شود). یک \lr{ridge} روی \lr{tokens} می‌تواند این دو را به هم ربط دهد بی آنکه \lr{EEG} اطلاعاتی داشته باشد. در آزمون روی داده‌ی کاملاً تصادفی این اتفاق افتاد: با جابه‌جایی‌های دلخواه، $r$ حدود \lr{۰٫۱} و «معنادار» بود. جابه‌جایی به اندازه‌ی مضرب گام پیش‌بینی، فاز این دوره را حفظ می‌کند؛ پس این اثر در \lr{null distribution} هم هست و فقط اطلاعاتی که به زمان واقعی بسته است شمرده می‌شود. ویژگی‌های \lr{band power} اصلاً موقعیت ندارند و بررسی تمیزتری هستند.

\begin{itemize}
\item \textbf{اثر واقعی است.} با \lr{tokens} میانگین $r$ برابر \lr{۰٫۰۷۳} و در \lr{null distribution} \lr{−۰٫۰۰۱} است؛ در ۱۰ شرکت‌کننده از ۱۲، $p\le0.05$ (از شانس کمتر از یک مورد انتظار می‌رود). با \lr{band power} میانگین $r$ برابر \lr{۰٫۰۴۰} و در ۶ نفر معنادار است.
\item \textbf{اثر بسیار کوچک است.} $r=0.07$ یعنی حدود \lr{۰٫۵} درصد واریانس باقی‌مانده. افزودن پیش‌بینی به \lr{K} به‌طور متوسط \lr{۰٫۱۰} واحد (با \lr{tokens}) و \lr{−۰٫۱۳} واحد (با \lr{band power}) به \lr{VAF} اضافه می‌کند. فقط در ۷ نفر (\lr{tokens}) و ۲ نفر (\lr{band power}) بهره‌ی \lr{VAF} هم مثبت و هم معنادار است.
\item \textbf{استثنا: \lr{P8}.} با \lr{tokens} \lr{۲٫۴} واحد بهره و $r$ تا \lr{۰٫۲۹} برای \lr{BR}، \lr{FD} و \lr{ED}. \lr{P8} ضعیف‌ترین \lr{K} را دارد (\lr{VAF} برابر ۵۸ درصد)، پس جای بیشتری برای توضیح دادن هست.
\item \textbf{ماژول \lr{fusion} همان را یاد گرفته که هست.} $r_{model}$ (اصلاح خودِ مدل در برابر باقی‌مانده) به‌طور متوسط \lr{۰٫۰۷۶} است، هم‌اندازه‌ی \lr{ridge}. \lr{fusion} جهت درست را پیدا کرده، اما چون اطلاعات این‌قدر کم است، بهینه‌سازی \lr{gate} را نزدیک صفر نگه داشته است.
\end{itemize}

\subsection{بحث}

\lr{EEG} در این داده سیگنال کوچک ولی سازگاری درباره‌ی آنچه سینماتیک توضیح نمی‌دهد دارد، اما ارزش پیش‌بینی آن کمتر از نیم واحد \lr{VAF} است. بنابراین $F\approx K$ نتیجه‌ی درست این داده است، نه نقص ماژول \lr{fusion}. حتی یک مرحله‌ی آموزش جداگانه فقط برای \lr{fusion} (با هر دو جریان منجمد و انتخاب بر اساس خطای \lr{F}) در بهترین حالت حدود \lr{۰٫۱} واحد به میانگین و حدود ۲ واحد به \lr{P8} اضافه می‌کرد؛ این هزینه‌ی آموزش دوباره‌ی ۱۲ نفر را توجیه نمی‌کند.

یک احتیاط: \lr{tokens} از \lr{band power} بهتر عمل کردند و \lr{tokens} برای پیش‌بینی \lr{EMG} آموزش دیده‌اند، پس مرحله‌ی کار را می‌دانند. بخشی از این سیگنال کوچک ممکن است زمان‌بندی حرکت یا \lr{artifact} حرکت در \lr{EEG} باشد و نه فرمان قشر حرکتی. $r$ معنادار در اینجا اطلاعات عصبی فراتر از سینماتیک را \emph{اثبات نمی‌کند}. آزمون بعدی می‌تواند همین \lr{ridge} روی باندها یا الکترودهایی باشد که کمتر در معرض \lr{artifact} حرکت‌اند.
